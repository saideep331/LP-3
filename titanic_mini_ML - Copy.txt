# Check Missing Values
print(titanic_data.isnull().any())
print(titanic_data.isnull().sum())

# Remove Cabin Column
titanic_data.drop('Cabin', axis=1, inplace=True)

titanic_data.head()

# Remove Missing Values
titanic_data.dropna(inplace=True)

print(titanic_data.isnull().sum())

# Convert Categorical Data into Dummy Variables
Sex = pd.get_dummies(titanic_data['Sex'])
Sex.head()

Passengerclass = pd.get_dummies(titanic_data['Pclass'])
Passengerclass.head()

Embark = pd.get_dummies(titanic_data['Embarked'])
Embark.head()

# Combine Dummy Variables
titanic_data = pd.concat(
    [titanic_data, Sex, Passengerclass, Embark], axis=1
)

titanic_data.head()

# Remove Unnecessary Columns
titanic_data.drop(
    ['PassengerId', 'Pclass', 'Name', 'Ticket',
     'Embarked', 'Sex'],
    axis=1, inplace=True
)

titanic_data.head()

# Logistic Regression Model
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression

X = titanic_data.drop(
    ['Survived', 'C', 'Q', 'S', 'male'], axis=1
)

Y = titanic_data.Survived

# Fix column name error
X.columns = X.columns.astype(str)

print(X.head())
print(X.shape[0])
print(Y.shape[0])

# Split Dataset into Training and Testing
x_train, x_test, y_train, y_test = train_test_split(
    X, Y, test_size=0.2, random_state=42
)

# Train Model
model = LogisticRegression(max_iter=1000)
model.fit(x_train, y_train)

# Predict Survival
prediction = model.predict(x_test)

print("Predictions:")
print(prediction)

# Display Test Data
x_test.head(3)

# Calculate Accuracy
from sklearn.metrics import accuracy_score

acc_logreg = round(
    accuracy_score(y_test, prediction) * 100, 2
)

print("Logistic Regression Accuracy:", acc_logreg, "%")

print("Titanic Survival Prediction Completed Successfully!")
