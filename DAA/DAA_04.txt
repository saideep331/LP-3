# A Dynamic Programming based Python Program for 0-1 Knapsack problem
# Returns the maximum value that can be put in a knapsack of capacity W

def knapsack(W, wt, val, n):
    # Making the 1D dp array (Space optimized implementation)
    dp = [0 for i in range(W + 1)]
    
    for i in range(1, n + 1):  # taking first i elements
        # starting from back, so that we also have data of previous computation when taking i-1 items
        for w in range(W, 0, -1):
            if wt[i - 1] <= w:
                # finding the maximum value
                dp[w] = max(dp[w], dp[w - wt[i - 1]] + val[i - 1])
                
    return dp[W]  # returning the maximum value of knapsack

# Driver code
if __name__ == "__main__":
    val = [60, 100, 120]
    wt = [10, 20, 30]
    W = 50
    n = len(val)
    print(knapsack(W, wt, val, n))

  #Time Complexity-> Pseudo Polynomial O(n*W)
  #Space Complexity- This code O(W) due to use of 1D DP array. Generally O(n*W)
